# Pimarchy headless setup
#
# Fill this in on the computer that flashed the card to set the Pi up without
# a screen or keyboard: put your value after the "=" of each setting you want,
# and leave the others empty. Leave the whole file as it is to set the Pi up at
# its screen. Save it as plain text.
#
# On its first boot the Pi reads this file, joins your Wi-Fi network, turns SSH
# on and creates your account, then blanks the passwords below and writes what
# it did to pimarchy-status.txt next to this file. A few minutes after power
# on, reach it from your computer with:
#
#   ssh USERNAME@HOSTNAME.local
#
# The Pi reads this file only until it has set up its account. A value is used
# as written; put it in "double quotes" to keep spaces at its start or end.

# --- Wi-Fi --------------------------------------------------------------------
# Your network's name and password. A name with no password joins an open
# network. Leave both empty on Ethernet.
WIFI_SSID=
WIFI_PASSWORD=
# The two-letter code of the country the Pi is in, such as US, GB or DE: the
# radio uses only the channels allowed there.
WIFI_COUNTRY=
# yes for a network that does not broadcast its name.
WIFI_HIDDEN=no

# --- Account ------------------------------------------------------------------
# Your username (lowercase letters, digits, - and _) and password, which root
# gets too. To keep the password itself off the card, leave PASSWORD empty and
# give PASSWORD_HASH its hash instead, which this makes:  openssl passwd -6
USERNAME=
PASSWORD=
PASSWORD_HASH=
# Your name and email, for git. Optional.
FULL_NAME=
EMAIL=

# --- Machine ------------------------------------------------------------------
# The Pi's name on the network (omarchy if empty), its timezone, such as
# Europe/London (UTC if empty), and its keyboard layout, such as us, uk, de or
# fr (us if empty).
HOSTNAME=
TIMEZONE=
KEYBOARD=

# --- SSH ----------------------------------------------------------------------
# SSH is on once this file sets up the account or gives a key. Paste your public
# key (the contents of ~/.ssh/id_ed25519.pub, say) after SSH_KEY=, and add an
# SSH_KEY= line for each other key. With a key, password logins are off unless
# SSH_PASSWORD_LOGIN is yes.
SSH_KEY=
SSH_PASSWORD_LOGIN=no
